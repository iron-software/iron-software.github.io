    public sealed class Direction : Enum

    Direction sets the reading direction text flows in. LeftToRight is the default for Latin scripts, and RightToLeft lays text out for languages such as Arabic and Hebrew. Set the value that matches the language of the content on a slide.

    var direction = Direction.LeftToRight;
